impl Stabilization {
    /// 生成 16×16 remap grid，复用你 OpenCV 的算法
    fn generate_remap_grid_fast(
        &self,
        timestamp_us: i64,
        grid_size: usize,  // 16
    ) -> Vec<f32> {
        let size = self.size;
        let out_w = self.output_size.0;
        let out_h = self.output_size.1;
        
        // 从 compute_params 获取内参
        let fx = self.compute_params.focal_length.0;
        let fy = self.compute_params.focal_length.1;
        let cx = self.compute_params.lens_center.0;
        let cy = self.compute_params.lens_center.1;
        let fov_scale = self.compute_params.fov_scale;
        
        let fx_z = fx * fov_scale;
        let fy_z = fy * fov_scale;
        
        // 获取四元数（从 smoothed_quaternions）
        let q_prev = self.get_quaternion_at_timestamp(timestamp_us - 1);  // 上一帧
        let q_curr = self.get_quaternion_at_timestamp(timestamp_us);        // 当前帧
        
        let mut map_data = vec![0.0f32; grid_size * grid_size * 2];
        
        for gy in 0..grid_size {
            let y = gy as f32 * (out_h as f32 - 1.0) / (grid_size as f32 - 1.0);
            let phase = (y / out_h as f32) * self.compute_params.rs_ratio;
            
            // RS 插值（行级四元数）
            let q_row = q_prev.slerp(phase, q_curr);
            
            // 转旋转矩阵（复用你的 Euler 角修正逻辑）
            let euler = quaternion_to_euler(&q_row);
            let euler_fixed = [-euler[0], euler[1], euler[2]];  // X 轴取反
            let r = euler_to_rotation_matrix(&euler_fixed);
            
            for gx in 0..grid_size {
                let x = gx as f32 * (out_w as f32 - 1.0) / (grid_size as f32 - 1.0);
                
                // 目标 -> 归一化平面（Zoom 内参）
                let xn = (x - cx) / fx_z;
                let yn = (y - cy) / fy_z;
                
                // 逆旋转
                let p = r.transpose() * Vector3::new(xn, yn, 1.0);
                
                // 源图像像素坐标（原始内参）
                let u = (p.x / p.z) * fx + cx;
                let v = (p.y / p.z) * fy + cy;
                
                let idx = (gy * grid_size + gx) * 2;
                map_data[idx] = u / size.0 as f32;      // 归一化到 [0,1]
                map_data[idx + 1] = v / size.1 as f32;
            }
        }
        
        map_data
    }

Kannala-Brandt 鱼眼模型
核心公式
plain
复制
输入：归一化平面坐标 (xn, yn, 1)
输出：畸变后的图像坐标 (u, v)

1. 角度 θ = atan(sqrt(xn² + yn²))
2. 径向畸变：r = θ + k1*θ³ + k2*θ⁵ + k3*θ⁷ + k4*θ⁹
3. 方向：xd = r * xn / sqrt(xn² + yn²)
         yd = r * yn / sqrt(xn² + yn²)
4. 像素坐标：u = xd * fx + cx
             v = yd * fy + cy
和 Brown-Conrady 的区别
表格
Brown-Conrady (普通广角)	Kannala-Brandt (鱼眼)
畸变变量	r² = x² + y²	θ = atan(r)
径向公式	多项式关于 r²	多项式关于 θ
适用镜头	普通广角、标准镜头	超广角、鱼眼
最大角度	< 100°	> 180°
Gyroflow WGSL 里的 distort_point()
从你的 WGSL 代码看，distort_point() 是 LENS_MODEL_FUNCTIONS 宏展开的。鱼眼模型的实现大概是：

fn distort_point(x: f32, y: f32, w: f32) -> vec2<f32> {
    let xn = x / w;
    let yn = y / w;
    let r = sqrt(xn*xn + yn*yn);
    
    if (r < 0.001) {
        return vec2<f32>(xn, yn);  // 中心附近无畸变
    }
    
    let theta = atan(r);
    let theta2 = theta * theta;
    let theta4 = theta2 * theta2;
    let theta6 = theta4 * theta2;
    let theta8 = theta4 * theta4;
    
    // 径向畸变：r_distorted = θ + k1*θ³ + k2*θ⁵ + k3*θ⁷ + k4*θ⁹
    let r_distorted = theta 
        + params.k1.x * theta2 * theta   // k1*θ³
        + params.k2.x * theta4 * theta   // k2*θ⁵
        + params.k3.x * theta6 * theta   // k3*θ⁷
        + params.k1.y * theta8 * theta;  // k4*θ⁹（注意：k4 存在 k1.y）
    
    let scale = r_distorted / r;
    
    return vec2<f32>(xn * scale, yn * scale);
}
注意：Gyroflow 的系数存储方式和 OpenCV 不同，k1 是 vec4<f32>，可能把 k1,k2,k3,k4 都存在里面。

        let r_fixed = euler_to_rotation_matrix(&euler_fixed);
        
        let mut map_data = vec![0.0f32; grid_size * grid_size * 2];
        
        for gy in 0..grid_size {
            let y = gy as f32 * (out_h as f32 - 1.0) / (grid_size as f32 - 1.0);
            
            // RS 行级插值
            let phase = (y / out_h as f32) * self.compute_params.rs_ratio;
            let q_row = self.get_quaternion_for_row(timestamp_us, phase);
            let r_row = quaternion_to_rotation_matrix(&q_row);
            
            for gx in 0..grid_size {
                let x = gx as f32 * (out_w as f32 - 1.0) / (grid_size as f32 - 1.0);
                
                // 1. 输出 -> 归一化平面（Zoom 内参）
                let xn = (x - cx) / fx_z;
                let yn = (y - cy) / fy_z;
                
                // 2. 逆旋转
                let p = r_row.transpose() * Vector3::new(xn, yn, 1.0);
                let xn = p.x / p.z;
                let yn = p.y / p.z;
                
                // 3. 加鱼眼畸变（Kannala-Brandt）
                let r_norm = (xn*xn + yn*yn).sqrt();
                let (xd, yd) = if r_norm < 0.001 {
                    (xn, yn)
                } else {
                    let theta = r_norm.atan();
                    let theta2 = theta * theta;
                    let theta4 = theta2 * theta2;
                    let theta6 = theta4 * theta2;
                    let theta8 = theta4 * theta4;
                    
                    let r_distorted = theta 
                        + k1 * theta2 * theta
                        + k2 * theta4 * theta
                        + k3 * theta6 * theta
                        + k4 * theta8 * theta;
                    
                    let scale = r_distorted / r_norm;
                    (xn * scale, yn * scale)
                };
                
                // 4. 源图像像素坐标
                let u = xd * fx + cx;
                let v = yd * fy + cy;
                
                let idx = (gy * grid_size + gx) * 2;
                map_data[idx] = u / size.0 as f32;
                map_data[idx + 1] = v / size.1 as f32;
            }
        }
        
        map_data
    }
}

pub struct WgpuWrapper {
    // ... 原有字段
    remap_texture: wgpu::Texture,
    remap_sampler: wgpu::Sampler,
    fast_pipeline: wgpu::ComputePipeline,  // 新的 fast shader
    fast_bind_group_layout: wgpu::BindGroupLayout,
}

impl WgpuWrapper {
    pub fn new(...) -> Self {
        // ... 原有初始化
        
        // 创建 remap texture（16×16 grid，可复用）
        let remap_texture = device.create_texture(&wgpu::TextureDescriptor {
            label: Some("remap_texture"),
            size: wgpu::Extent3d {
                width: 16,  // grid_size
                height: 16,
                depth_or_array_layers: 1,
            },
            mip_level_count: 1,
            sample_count: 1,
            dimension: wgpu::TextureDimension::D2,
            format: wgpu::TextureFormat::Rg32Float,  // RG32F
            usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
            view_formats: &[],
        });
        
        // ... 创建 fast_pipeline 和 fast_bind_group_layout
    }
    
    pub fn undistort_image_fast(
        &self,
        buffers: &mut Buffers,
        itm: &FrameTransform,
        drawing_buffer: Option<&[u8]>,
        remap_data: &[f32],  // CPU 预计算的 grid
    ) {
        // 1. 上传 remap 数据
        let remap_size = (16u32, 16u32);
        self.queue.write_texture(
            wgpu::ImageCopyTexture {
                texture: &self.remap_texture,
                mip_level: 0,
                origin: wgpu::Origin3d::ZERO,
                aspect: wgpu::TextureAspect::All,
            },
            bytemuck::cast_slice(remap_data),
            wgpu::ImageDataLayout {
                offset: 0,
                bytes_per_row: Some(remap_size.0 * 8),  // 2 * f32 = 8 bytes
                rows_per_image: Some(remap_size.1),
            },
            wgpu::Extent3d {
                width: remap_size.0,
                height: remap_size.1,
                depth_or_array_layers: 1,
            },
        );
        
        // 2. 创建 fast bind group
        let fast_bind_group = self.device.create_bind_group(&wgpu::BindGroupDescriptor {
            label: Some("fast_bind_group"),
            layout: &self.fast_bind_group_layout,
            entries: &[
                // @binding(0) params
                wgpu::BindGroupEntry {
                    binding: 0,
                    resource: wgpu::BindingResource::Buffer(...),
                },
                // @binding(5) input_texture
                wgpu::BindGroupEntry {
                    binding: 5,
                    resource: wgpu::BindingResource::TextureView(...),
                },
                // @binding(6) output_buffer
                wgpu::BindGroupEntry {
                    binding: 6,
                    resource: wgpu::BindingResource::Buffer(...),
                },
                // @binding(7) remap_tex
                wgpu::BindGroupEntry {
                    binding: 7,
                    resource: wgpu::BindingResource::TextureView(
                        &self.remap_texture.create_view(&wgpu::TextureViewDescriptor::default())
                    ),
                },
                // @binding(8) remap_sampler
                wgpu::BindGroupEntry {
                    binding: 8,
                    resource: wgpu::BindingResource::Sampler(&self.remap_sampler),
                },
            ],
        });
        
        // 3. 派发 compute shader
        let mut encoder = self.device.create_command_encoder(...);
        {
            let mut compute_pass = encoder.begin_compute_pass(...);
            compute_pass.set_pipeline(&self.fast_pipeline);
            compute_pass.set_bind_group(0, &fast_bind_group, &[]);
            
            let workgroups_x = (itm.kernel_params.output_width as u32 + 15) / 16;
            let workgroups_y = (itm.kernel_params.output_height as u32 + 15) / 16;
            compute_pass.dispatch_workgroups(workgroups_x, workgroups_y, 1);
        }
        
        // 4. 提交
        self.queue.submit(std::iter::once(encoder.finish()));
    }
}

Gyroflow 当前 shader（核显 i7 = 8ms）

复制
每个像素：
  undistort_coord() × 1-3 次
    └── rotate_and_distort()
          ├── 3×3 矩阵乘法（9 mul + 6 add）
          ├── distort_point() — 鱼眼：atan + sqrt + 多项式
          ├── mesh 双变量样条（有条件）
          ├── 数字镜头（有条件）
          └── FocalPlaneDistortion（有条件）
    sample_input_at()
      ├── EWA: affine_bbox + clamped_ellipse（三角函数）
      └── 双重循环 5×5 ~ 9×9 邻域采样
4K (3840×2160) ≈ 830万像素
每像素 ~500-2000 条 ALU 指令
EWA 额外 25-81 次纹理采样
核显 24-32 EU，每个 EU 8 线程 → 总并行度 ~256 线程
8ms 已经是核显极限
